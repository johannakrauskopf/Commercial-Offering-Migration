WITH slot AS (
  SELECT m.job_slot_id, MAX(m.job_id) AS job_id, BOOL_OR(m.is_pilot) AS is_pilot,
         MAX(m.paid_credit_used_eur + m.goodwill_credit_used_eur) AS credit,
         MIN(m.job_slot_redeemed_dt) AS redeemed_dt, MAX(m.gtm_motion_current_company) AS motion
  FROM rds.rep_job_metrics m WHERE m.product_type='Hire' GROUP BY m.job_slot_id ),
enr AS (
  SELECT s.*, js.company_id,
         (js.hire_v2_offer='true') AS c_cur,
         (s.redeemed_dt >= DATE '2025-05-01' AND s.redeemed_dt <= DATE '2026-04-30') AS c_base
  FROM slot s JOIN backend.job_slots js ON js.id=s.job_slot_id
  LEFT JOIN salesforce.sf_account sa ON sa.txtcompanyid__c=js.company_id
  WHERE COALESCE(s.motion,'')<>'Agency Sales' AND COALESCE(sa.test_account__c,false)=false ),
pc AS (
  SELECT 'base' AS coh, company_id, is_pilot, COUNT(DISTINCT job_id) AS jobs,
         (DATEDIFF(day, MIN(redeemed_dt), DATE '2026-04-30')+1)/30.44 AS tenure_m
  FROM enr WHERE c_base GROUP BY company_id, is_pilot
  UNION ALL
  SELECT 'cur', company_id, is_pilot, COUNT(DISTINCT job_id) AS jobs,
         (DATEDIFF(day, MIN(redeemed_dt), CURRENT_DATE)+1)/30.44 AS tenure_m
  FROM enr WHERE c_cur GROUP BY company_id, is_pilot ),
job_seg AS (
  SELECT job_id, BOOL_OR(is_pilot) AS is_pilot, BOOL_OR(c_cur) AS is_hv2,
         MAX(CASE WHEN c_base THEN 1 ELSE 0 END) AS in_base
  FROM enr GROUP BY job_id ),
dj AS ( SELECT id AS job_id, CASE WHEN runtime ~ '^[0-9]+$' THEN runtime::int END AS rd FROM backend.dmsraw_jobs ),
jr AS ( SELECT jsg.is_pilot, jsg.is_hv2, jsg.in_base, dj.rd AS runtime_days FROM job_seg jsg JOIN dj ON dj.job_id=jsg.job_id ),
cjp AS ( SELECT e.job_id, BOOL_OR(e.is_pilot) AS is_pilot, SUM(e.credit) AS credit, COUNT(*) AS slots
         FROM enr e WHERE e.c_cur GROUP BY e.job_id ),
-- Refund: BOOKED universe (backend.job_slots), by created_at; Agency via company motion
ag AS ( SELECT DISTINCT js.company_id FROM rds.rep_job_metrics m JOIN backend.job_slots js ON js.id=m.job_slot_id
        WHERE m.gtm_motion_current_company='Agency Sales' ),
rfx AS (
  SELECT (s.refunded_at IS NOT NULL) AS refunded,
         (s.hire_v2_offer='true') AS c_cur,
         (s.created_at::date BETWEEN DATE '2025-05-01' AND DATE '2026-04-30' AND COALESCE(s.hire_v2_offer,'')<>'true') AS c_base
  FROM backend.job_slots s
  LEFT JOIN salesforce.sf_account sa ON sa.txtcompanyid__c=s.company_id
  LEFT JOIN ag ON ag.company_id=s.company_id
  WHERE COALESCE(sa.test_account__c,false)=false AND ag.company_id IS NULL ),
eur AS (
  SELECT
   ROUND(SUM(CASE WHEN c_cur  AND is_pilot     THEN credit END)/NULLIF(COUNT(DISTINCT CASE WHEN c_cur  AND is_pilot     THEN job_slot_id END),0)) AS slot_pilot_cur,
   ROUND(SUM(CASE WHEN c_base AND is_pilot     THEN credit END)/NULLIF(COUNT(DISTINCT CASE WHEN c_base AND is_pilot     THEN job_slot_id END),0)) AS slot_pilot_base,
   ROUND(SUM(CASE WHEN c_cur  AND NOT is_pilot THEN credit END)/NULLIF(COUNT(DISTINCT CASE WHEN c_cur  AND NOT is_pilot THEN job_slot_id END),0)) AS slot_nonpilot_cur,
   ROUND(SUM(CASE WHEN c_base AND NOT is_pilot THEN credit END)/NULLIF(COUNT(DISTINCT CASE WHEN c_base AND NOT is_pilot THEN job_slot_id END),0)) AS slot_nonpilot_base,
   ROUND(SUM(CASE WHEN c_base AND is_pilot     THEN credit END)/NULLIF(COUNT(DISTINCT CASE WHEN c_base AND is_pilot     THEN job_id END),0)) AS job_pilot_base,
   ROUND(SUM(CASE WHEN c_base AND NOT is_pilot THEN credit END)/NULLIF(COUNT(DISTINCT CASE WHEN c_base AND NOT is_pilot THEN job_id END),0)) AS job_nonpilot_base
  FROM enr ),
jbcur AS (
  SELECT
   ROUND(AVG(CASE WHEN cjp.is_pilot=true  THEN cjp.credit * dj.rd::decimal/(30.0*cjp.slots) END)) AS job_pilot_cur,
   ROUND(AVG(CASE WHEN cjp.is_pilot=false THEN cjp.credit * dj.rd::decimal/(30.0*cjp.slots) END)) AS job_nonpilot_cur
  FROM cjp JOIN dj ON dj.job_id=cjp.job_id WHERE cjp.is_pilot IS NOT NULL AND dj.rd IS NOT NULL ),
jpm AS (
  SELECT
   ROUND(SUM(CASE WHEN coh='cur'  AND is_pilot=true  THEN jobs END)::decimal/NULLIF(SUM(CASE WHEN coh='cur'  AND is_pilot=true  THEN tenure_m END),0),3) AS jpm_pilot_cur,
   ROUND(SUM(CASE WHEN coh='base' AND is_pilot=true  THEN jobs END)::decimal/NULLIF(SUM(CASE WHEN coh='base' AND is_pilot=true  THEN tenure_m END),0),3) AS jpm_pilot_base,
   ROUND(SUM(CASE WHEN coh='cur'  AND is_pilot=false THEN jobs END)::decimal/NULLIF(SUM(CASE WHEN coh='cur'  AND is_pilot=false THEN tenure_m END),0),3) AS jpm_nonpilot_cur,
   ROUND(SUM(CASE WHEN coh='base' AND is_pilot=false THEN jobs END)::decimal/NULLIF(SUM(CASE WHEN coh='base' AND is_pilot=false THEN tenure_m END),0),3) AS jpm_nonpilot_base
  FROM pc WHERE is_pilot IS NOT NULL ),
rt AS (
  SELECT
   ROUND(AVG(CASE WHEN is_hv2   AND is_pilot=true  THEN runtime_days END)/30.44,3) AS rt_pilot_cur,
   ROUND(AVG(CASE WHEN in_base=1 AND is_pilot=true  THEN runtime_days END)/30.44,3) AS rt_pilot_base,
   ROUND(AVG(CASE WHEN is_hv2   AND is_pilot=false THEN runtime_days END)/30.44,3) AS rt_nonpilot_cur,
   ROUND(AVG(CASE WHEN in_base=1 AND is_pilot=false THEN runtime_days END)/30.44,3) AS rt_nonpilot_base
  FROM jr WHERE is_pilot IS NOT NULL ),
rf AS (
  SELECT
   ROUND(100.0*SUM(CASE WHEN c_cur  AND refunded THEN 1 ELSE 0 END)/NULLIF(SUM(CASE WHEN c_cur  THEN 1 ELSE 0 END),0),3) AS rf_cur,
   ROUND(100.0*SUM(CASE WHEN c_base AND refunded THEN 1 ELSE 0 END)/NULLIF(SUM(CASE WHEN c_base THEN 1 ELSE 0 END),0),3) AS rf_base
  FROM rfx )
SELECT eur.*, jbcur.*, jpm.*, rt.*, rf.* FROM eur CROSS JOIN jbcur CROSS JOIN jpm CROSS JOIN rt CROSS JOIN rf;
